cmake_minimum_required(VERSION 3.16)
project(rmw_cpp_bench LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(rmw_bench node.cpp)
target_compile_features(rmw_bench PRIVATE cxx_std_17)
target_compile_options(rmw_bench PRIVATE -O3 -Wall -Wextra)
ament_target_dependencies(rmw_bench rclcpp std_msgs)
