cmake_minimum_required(VERSION 3.20)
project(CompanyWebView2UnityPilot VERSION 58.5.2 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Experimental WebView2 Unity pilot is Windows x64 only")
endif()
if(NOT WEBVIEW2_SDK_ROOT)
  message(FATAL_ERROR "Specify pinned WEBVIEW2_SDK_ROOT")
endif()
find_path(WV2_INC NAMES WebView2.h PATHS "${WEBVIEW2_SDK_ROOT}/build/native/include" NO_DEFAULT_PATH REQUIRED)
find_library(WV2_LOADER NAMES WebView2LoaderStatic PATHS "${WEBVIEW2_SDK_ROOT}/build/native/x64" NO_DEFAULT_PATH REQUIRED)
add_library(CompanyWebView2OverlayPilot SHARED WebView2UnityPilot.cpp)
target_compile_features(CompanyWebView2OverlayPilot PRIVATE cxx_std_17)
target_compile_definitions(CompanyWebView2OverlayPilot PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(CompanyWebView2OverlayPilot PRIVATE "${WV2_INC}")
target_link_libraries(CompanyWebView2OverlayPilot PRIVATE "${WV2_LOADER}" ole32 user32)
if(MSVC)
  target_compile_options(CompanyWebView2OverlayPilot PRIVATE /W4 /EHsc)
endif()
