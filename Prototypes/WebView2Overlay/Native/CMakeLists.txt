cmake_minimum_required(VERSION 3.20)
project(CompanyWebView2Overlay VERSION 58.5.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 only")
endif()
if(NOT WEBVIEW2_SDK_ROOT)
 message(FATAL_ERROR "Specify WEBVIEW2_SDK_ROOT")
endif()
find_path(WEBVIEW2_INCLUDE WebView2.h PATHS "${WEBVIEW2_SDK_ROOT}/build/native/include" NO_DEFAULT_PATH REQUIRED)
find_library(WEBVIEW2_LOADER WebView2LoaderStatic PATHS "${WEBVIEW2_SDK_ROOT}/build/native/x64" NO_DEFAULT_PATH REQUIRED)
add_executable(CompanyWebView2OverlayProbe WIN32 OverlayProbe.cpp)
target_compile_features(CompanyWebView2OverlayProbe PRIVATE cxx_std_17)
target_compile_definitions(CompanyWebView2OverlayProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(CompanyWebView2OverlayProbe PRIVATE "${WEBVIEW2_INCLUDE}")
target_link_libraries(CompanyWebView2OverlayProbe PRIVATE "${WEBVIEW2_LOADER}" ole32 user32)
if(MSVC)
 target_compile_options(CompanyWebView2OverlayProbe PRIVATE /W4 /EHsc)
endif()
