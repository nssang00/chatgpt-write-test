cmake_minimum_required(VERSION 3.20)
project(CompanyWebView2UnifiedPreview VERSION 58.6.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Unified preview native bridge requires Windows x64")
endif()
if(NOT WEBVIEW2_SDK_ROOT)
  message(FATAL_ERROR "WEBVIEW2_SDK_ROOT must be the pinned NuGet package")
endif()
set(PILOT_SRC "${CMAKE_CURRENT_SOURCE_DIR}/../../../Prototypes/WebView2Overlay/OneClick-v58.5.2/Native")
find_path(WV2_INCLUDE NAMES WebView2.h PATHS "${WEBVIEW2_SDK_ROOT}/build/native/include" NO_DEFAULT_PATH REQUIRED)
find_library(WV2_LOADER NAMES WebView2LoaderStatic PATHS "${WEBVIEW2_SDK_ROOT}/build/native/x64" NO_DEFAULT_PATH REQUIRED)
add_library(CompanyWebView2UnifiedPreview SHARED "${PILOT_SRC}/WebView2UnityPilot.cpp")
target_compile_features(CompanyWebView2UnifiedPreview PRIVATE cxx_std_17)
target_compile_definitions(CompanyWebView2UnifiedPreview PRIVATE UNIFIED_PREVIEW UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(CompanyWebView2UnifiedPreview PRIVATE "${WV2_INCLUDE}" "${PILOT_SRC}")
target_link_libraries(CompanyWebView2UnifiedPreview PRIVATE "${WV2_LOADER}" ole32 user32)
if(MSVC)
  target_compile_options(CompanyWebView2UnifiedPreview PRIVATE /W4 /EHsc /utf-8)
endif()
enable_testing()
add_executable(UnifiedNativeContractTest UnifiedNativeContractTest.cpp)
target_compile_features(UnifiedNativeContractTest PRIVATE cxx_std_17)
target_compile_definitions(UnifiedNativeContractTest PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(UnifiedNativeContractTest PRIVATE /W4 /EHsc /utf-8)
endif()
add_test(NAME unified-native-exports COMMAND UnifiedNativeContractTest "$<TARGET_FILE:CompanyWebView2UnifiedPreview>")
