cmake_minimum_required(VERSION 3.20)
project(CompanyWebView2SecureV8 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Requires MSVC Win64 with official Microsoft WebView2 SDK")
endif()
if(NOT DEFINED WEBVIEW2_SDK_ROOT)
    message(FATAL_ERROR "Set WEBVIEW2_SDK_ROOT to unpacked Microsoft.Web.WebView2 NuGet package")
endif()
set(WV2_INCLUDE "${WEBVIEW2_SDK_ROOT}/build/native/include")
set(WV2_LIB "${WEBVIEW2_SDK_ROOT}/build/native/x64/WebView2LoaderStatic.lib")
if(NOT EXISTS "${WV2_INCLUDE}/WebView2.h" OR NOT EXISTS "${WV2_LIB}")
    message(FATAL_ERROR "Missing genuine WebView2.h or WebView2LoaderStatic.lib")
endif()
add_library(CompanyWebView2SecureV8 SHARED CwvWebView2Native.cpp)
target_compile_features(CompanyWebView2SecureV8 PRIVATE cxx_std_17)
target_include_directories(CompanyWebView2SecureV8 PRIVATE "${WV2_INCLUDE}")
target_link_libraries(CompanyWebView2SecureV8 PRIVATE "${WV2_LIB}" ole32 user32)
target_compile_options(CompanyWebView2SecureV8 PRIVATE /W4 /permissive- /EHsc)
set_target_properties(CompanyWebView2SecureV8 PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
